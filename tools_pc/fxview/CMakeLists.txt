# fxview on ge_render (issue #144): behind GE_TOOLS. PC: build-*/tools_pc/fxview/fxview[.exe]; Switch: fxview.nro.
# Clean include paths (imgui.h cannot see the game's include/, D66): the tool talks to the renderer through ge_render.h only.

# What the viewer needs to find the sprites in the user's ROM, taken from the repo at configure time so the Switch build needs no
# repo files at run time: the US image list (offsets only), the global image table's ROM address (port/src/romassets_u.s) and the
# sprite tables' offsets in it (src/game/image_bank.c). The ROM content itself is read at run time and never written anywhere.
set(_fx_inc "${CMAKE_CURRENT_BINARY_DIR}/fx_romdata.inc")
file(READ "${CMAKE_SOURCE_DIR}/imagelist.u.csv" _fx_csv)
file(READ "${CMAKE_SOURCE_DIR}/port/src/romassets_u.s" _fx_as)
file(READ "${CMAKE_SOURCE_DIR}/src/game/image_bank.c" _fx_bank)
string(REGEX MATCH "_GlobalimagetableSegmentRomStart, *0x([0-9A-Fa-f]+)" _fx_m "${_fx_as}")
if(NOT CMAKE_MATCH_1)
  message(FATAL_ERROR "fxview: no _GlobalimagetableSegmentRomStart in port/src/romassets_u.s")
endif()
set(_fx_seg "0x${CMAKE_MATCH_1}ull")
set(_fx_offs "")
foreach(_n impactimages explosion_smokeimages scattered_explosions flareimage1 flareimage2 flareimage3 flareimage4 flareimage5 ammo9mmimage)
  string(REGEX MATCH "g_pc_gimg_off_s_${_n} *= *0x([0-9A-Fa-f]+)" _fx_m "${_fx_bank}")
  if(NOT CMAKE_MATCH_1)
    message(FATAL_ERROR "fxview: no g_pc_gimg_off_s_${_n} in src/game/image_bank.c")
  endif()
  string(APPEND _fx_offs "0x${CMAKE_MATCH_1}u, ")
endforeach()
file(WRITE "${_fx_inc}" "// generated by tools_pc/fxview/CMakeLists.txt\nstatic const unsigned long long kSegU = ${_fx_seg};\nstatic const unsigned kOffU[9] = { ${_fx_offs} };\nstatic const char kImageListU[] = R\"IMGL(")
file(APPEND "${_fx_inc}" "${_fx_csv}")
file(APPEND "${_fx_inc}" ")IMGL\";\n")
# The file table (offset,size,path: the guards' weapon models carry the muzzle flashes) and the texture names with their hit type
# (assets/images.def: the texture sheet, SMOKE_0 for the explosions): lists of names and offsets from the repo, nothing from the ROM.
file(READ "${CMAKE_SOURCE_DIR}/scripts/filelist.u.csv" _fx_files)
file(APPEND "${_fx_inc}" "static const char kFileListU[] = R\"FLST(${_fx_files})FLST\";\n")
file(READ "${CMAKE_SOURCE_DIR}/assets/images.def" _fx_def)
string(REGEX MATCHALL "IMAGE\\([^)]*\\)" _fx_imgs "${_fx_def}")
set(_fx_names "")
foreach(_i IN LISTS _fx_imgs)
  string(REGEX REPLACE "IMAGE\\(([^,]*),[^,]*,[^,]*, *([A-Za-z_]*),.*" "\\1,\\2" _l "${_i}")
  string(APPEND _fx_names "${_l}\n")
endforeach()
file(APPEND "${_fx_inc}" "static const char kTexNames[] = R\"TNAM(${_fx_names})TNAM\";\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/imagelist.u.csv" "${CMAKE_SOURCE_DIR}/port/src/romassets_u.s"
                                                               "${CMAKE_SOURCE_DIR}/src/game/image_bank.c" "${CMAKE_SOURCE_DIR}/scripts/filelist.u.csv"
                                                               "${CMAKE_SOURCE_DIR}/assets/images.def")

set(_mv "${CMAKE_SOURCE_DIR}/tools_pc/modelview/pc")   # modelview's texture decoder and inflate (reused as is)
add_executable(fxview ge/fxview.cpp ge/fx_rom.cpp ge/fx_tex.cpp "${_mv}/mv_tex.cpp" "${_mv}/mv_zlib.cpp")
set_property(TARGET fxview PROPERTY INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/port/third_party/imgui" "${CMAKE_SOURCE_DIR}/port/ge_render/include"
                                                         "${CMAKE_SOURCE_DIR}/port/include" "${_mv}" "${CMAKE_CURRENT_BINARY_DIR}"
                                                         "${CMAKE_SOURCE_DIR}/tools_pc/fxview/sea" "${SDL2_INCLUDE_DIR}")
set_property(TARGET fxview PROPERTY COMPILE_OPTIONS ${_imgui_opts})
set_property(TARGET fxview PROPERTY COMPILE_DEFINITIONS "")
set_property(TARGET fxview PROPERTY CXX_STANDARD 20)
if(GE_SWITCH)
  set_property(TARGET fxview PROPERTY POSITION_INDEPENDENT_CODE OFF)
  target_compile_options(fxview PRIVATE -fPIE)
endif()
if(NOT GE_SWITCH)   # PC: the effect shaders' hot reload (dkshgen compiled in, an external glslangValidator for Vulkan)
  target_sources(fxview PRIVATE ge/fx_reload.cpp)
  target_include_directories(fxview PRIVATE "${CMAKE_SOURCE_DIR}/port/ge_render/src")
endif()
target_link_libraries(fxview fxview_sea ge_render_imgui ge_imgui ge_render)   # fxview_sea: --mode sea (the GE+ sky/sea/water)
if(TARGET fxview_sea_pack)
  set_property(TARGET fxview APPEND PROPERTY COMPILE_DEFINITIONS SEA_HAVE_PACK=1)
endif()
if(GE_SWITCH)
  nx_generate_nacp(fxview.nacp NAME "fxview" AUTHOR "Project Dam" VERSION "0")
  nx_create_nro(fxview NACP fxview.nacp ICON "${NX_DEFAULT_ICON}")
endif()
